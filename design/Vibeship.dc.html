src/routes/book.ts
import { Hono } from 'hono';
import { zValidator } from '@hono/zod-validator';
import { bookingSchema } from '../schema';

// Резервация: пише в D1 и пуска SMS в Queue
export const book = new Hono<{ Bindings: Env }>();

book.post('/api/book', zValidator('json', bookingSchema), async (c) => {
  const { barberId, slot, name, phone } = c.req.valid('json');

  const taken = await c.env.DB
    .prepare('SELECT 1 FROM appointments WHERE barber_id = ? AND slot = ?')
    .bind(barberId, slot)
    .first();

  if (taken) return c.json({ error: 'Часът е зает' }, 409);

  await c.env.DB
    .prepare('INSERT INTO appointments (barber_id, slot, name, phone) VALUES (?, ?, ?, ?)')
    .bind(barberId, slot, name, phone)
    .run();

  await c.env.SMS_QUEUE.send({ phone, slot, kind: 'reminder' });

  return c.json({ ok: true }, 201);
});
